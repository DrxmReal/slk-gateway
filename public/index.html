from openai import OpenAI

# Khởi tạo client với API Key Discord của bạn
client = OpenAI(
    base_url="$GATEWAY_URL",
    api_key="$USER_API_KEY"
)

response = client.chat.completions.create(
    model="DeepSeek-V4-Flash",  # Free models: DeepSeek-V4-Flash, auto, glm-5.3-flash, kimi-k3, longcat-2.5
    messages=[
        {"role": "user", "content": "Xin chào, hãy giới thiệu về bạn."}
    ],
    stream=True
)

for chunk in response:
    content = chunk.choices[0].delta.content or ""
    print(content, end="", flush=True)

import OpenAI from 'openai';

const openai = new OpenAI({
  baseURL: '$GATEWAY_URL',
  apiKey: '$USER_API_KEY',
});

async function main() {
  const stream = await openai.chat.completions.create({
    model: 'DeepSeek-V4-Flash',
    messages: [{ role: 'user', content: 'Giải thích cơ chế Proxy API?' }],
    stream: true,
  });

  for await (const chunk of stream) {
    process.stdout.write(chunk.choices[0]?.delta?.content || '');
  }
}

main();
